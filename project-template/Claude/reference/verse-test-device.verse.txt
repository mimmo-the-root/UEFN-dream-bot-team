# Centralized logger — rule 2
@editable DebugLoggingEnabled : logic = true
Log(LogMessage : string):void=
    if (DebugLoggingEnabled?):
        Print(LogMessage)

# Test assertions
ExpectEqual<T>(Expected : T, Actual : T, TestName : string):void=
    if (Expected = Actual):
        Log("[KIT-TEST] PASS {TestName}: expected {Expected}, got {Actual}")
    else:
        Log("[KIT-TEST] FAIL {TestName}: expected {Expected}, got {Actual}")

ExpectTrue(Condition : logic, TestName : string):void=
    if (Condition):
        Log("[KIT-TEST] PASS {TestName}: condition is true")
    else:
        Log("[KIT-TEST] FAIL {TestName}: condition is false")

ExpectNear(Expected : float, Actual : float, Tolerance : float, TestName : string):void=
    Diff : float = Abs(Expected - Actual)
    if (Diff <= Tolerance):
        Log("[KIT-TEST] PASS {TestName}: expected ~{Expected}, got {Actual} (diff {Diff} <= tol {Tolerance})")
    else:
        Log("[KIT-TEST] FAIL {TestName}: expected ~{Expected}, got {Actual} (diff {Diff} > tol {Tolerance})")
